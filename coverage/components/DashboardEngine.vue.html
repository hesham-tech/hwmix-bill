// حساب المكونات المضافة ولكن حالتها visible = false (المخفية)

  if (!dashboardStore.layout || !dashboardStore.layout.widgetInstances) return [];
  return dashboardStore.layout.widgetInstances.filter(wi => wi.visible === false);
});
